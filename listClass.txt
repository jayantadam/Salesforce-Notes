     List<Integer> nums = new List<Integer>();

         nums.addAll(new List<Integer>{30, 40});

       	system.debug(nums[1]); // DEBUG|(30)

            nums[0] = 100;
            system.debug(nums); // DEBUG|(100, 30, 40)
    //Remove item (by index):
        nums.remove(0);     // empty

        system.debug(nums.size()); // 2
    //isEmpty
        system.debug(nums.isEmpty());   //false
    //contains
        system.debug(nums.contains(30));   // true

        List<integer>copy = nums.clone();
